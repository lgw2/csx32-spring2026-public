\headers{CSCI 432/532}{Problem Session 7b}{Spring 2026}


\begin{enumerate}
\item 
Let $M$ be an arbitrary Turing machine. Prove the following claims.
\begin{enumerate}[(a)]
    \item There is a Turing machine $M^R$ such that $$\mathsc{Accept}(M^R) =
    \mathsc{Reject}(M)$$ and $$\mathsc{Reject}(M^R)=\mathsc{Accept}(M).$$
    \item There is a Turing machine $M^A$ such that $$\mathsc{Accept}(M^A) =
    \mathsc{Accept}(M)$$ and $$\mathsc{Reject}(M^A)=\emptyset.$$
    \item There is a Turing machine $M^H$ such that $$\mathsc{Accept}(M^H) =
    \mathsc{Halt}(M)$$ and $$\mathsc{Reject}(M^H)=\emptyset.$$
\end{enumerate}

\item Let $\mathsc{SelfAccept} = \{\langle M \rangle: M \text{ accepts } \langle
M \rangle \}$. Prove that $\mathsc{SelfAccept}$ is undecidable.

\item Let $\mathsc{Accept} = \{\langle M, w \rangle: M \text{ accepts } w\}$. Prove that $\mathsc{Accept}$ is undecidable.


\end{enumerate}

\begin{enumerate}\itemsep 4ex

\item Prove that each of the following languages is undecidable.
\begin{enumerate}[(a)]
    \item $\mathsc{AlwayAccept} = \{\langle M \rangle: \mathsc{Accept}(M) = \Sigma^*\}$
    \item $\mathsc{AlwayReject} = \{\langle M \rangle: \mathsc{Reject}(M) = \Sigma^*\}$
    \item $\mathsc{AlwayHalt} = \{\langle M \rangle: \mathsc{Halt}(M) = \Sigma^*\}$
    \item $\mathsc{AlwayDiverge} = \{\langle M \rangle: \mathsc{Diverge}(M) = \Sigma^*\}$
\end{enumerate}


\end{enumerate}
